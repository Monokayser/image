CI Workflow .github/workflows/ci.yml 
on:
  push:
    branches: ["**"]
  pull_request:

steps:
  checkout
  setup-java 21
  setup-node 20
  npm install
  npm run test
  npm run build
  mvn test
Docker Publish Workflow .github/workflows/docker-publish.yml 
on:
  push:
    branches: ["main"]
    tags: ["v*"]

steps:
  checkout
  setup-buildx
  docker login
  extract metadata
  build and push image
